# Phase-0 bring-up config for the Pi Zero 2 W.
#
# References (see docs/REAL_HW_BRINGUP.md for the verified-facts section):
#   - raspberrypi.com/documentation/computers/config_txt.html
#   - raspberrypi/firmware/boot/overlays/README (master)
#   - raspberrypi/tools/armstubs/armstub8.S

# 64-bit kernel. Selects the AArch64 armstub which eret's to EL2h before
# branching to kernel8.img at 0x80000. Without this we land at EL3 or
# the 32-bit armstub.
arm_64bit=1

# Use the cut-down GPU firmware (start_cd.elf + fixup_cd.dat).
# Per the config.txt docs: "Cut-down firmware (start*cd.elf and
# fixup*cd.dat) cannot be selected this way [via start_file=] - the
# system will fail to boot. The only way to enable the cut-down
# firmware is to specify gpu_mem=16." We ship start_cd.elf, so this
# line is mandatory — otherwise the firmware looks for start.elf
# (which isn't on the SD) and halts after reading config.txt.
gpu_mem=16

# Our binary (built by scripts/build-sd.sh from the pi-probe [[bin]]).
kernel=kernel8.img

# Enable a serial console on GPIO 14/15 (header pins 8 and 10).
enable_uart=1

# Also enable firmware-side UART logging from bootcode.bin and start.elf.
# Useful as an early checkpoint: if no bytes appear on the wire at all,
# the problem is upstream of our kernel (SD layout, GPIO ALT, baud).
uart_2ndstage=1

# CRITICAL: PL011 (UART0) is wired to the onboard Bluetooth chip by
# default on the Pi Zero 2 W; the GPIO header carries the mini-UART
# (UART1) instead. Our code drives PL011 at 0x3F20_1000, so we route it
# to GPIO 14/15 by disabling BT. Quote from raspberrypi/firmware
# overlays/README: "Disable onboard Bluetooth on Bluetooth-capable
# Raspberry Pis. On Pis prior to Pi 5 this restores UART0/ttyAMA0 over
# GPIOs 14 & 15."
dtoverlay=disable-bt

# Suppress the firmware's on-screen warning icons (the lightning bolt
# for under-voltage, the thermometer for over-temperature, and the
# multicolour test square). They draw on a separate overlay layer
# above our framebuffer and would otherwise show through any
# rendered content. Setting to 1 disables only the *icons* — the
# underlying protection (thermal throttling, etc.) still applies.
# Set to 0 (or delete this line) to see the icons again if you ever
# want to diagnose a real power / thermal issue.
avoid_warnings=1

# Disable HDMI overscan padding. Default firmware behaviour assumes
# the display has consumer-TV overscan and trims a margin (visible
# as a fixed-colour bar at the edges, typically top/bottom). On
# monitors with no real overscan that margin is just lost pixels —
# and on the Pi Zero 2 W with 1360x768 EDID it looked like an
# 8-16px white stripe across the top of our framebuffer.
disable_overscan=1

# HDMI link-stability knobs. Walter saw intermittent flicker that
# unplug-replug of the cable would sometimes fix — classic
# signature of a marginal HDMI lock at boot. The monitor under
# test is a small Pi-targeted panel reporting DMT 39 (1360x768
# @60); both the cable and the panel's HPD/EDID handling can be
# borderline at that timing.
#
#   hdmi_drive=2          force HDMI mode (with audio) — some
#                         marginal panels otherwise drop to DVI
#                         and lose sync.
#   hdmi_force_hotplug=1  ignore HPD; drive HDMI unconditionally.
#                         Defeats noisy hotplug-detect signalling
#                         on small panels that briefly de-assert
#                         HPD after their own internal events.
#   config_hdmi_boost=7   stronger differential signal (default 5,
#                         max 11). Improves margin on borderline
#                         cables / connectors without going crazy.
hdmi_drive=2
hdmi_force_hotplug=1
config_hdmi_boost=7

# Force CEA 720p60 (1280x720 @60 Hz, mode 4 in HDMI CEA group).
# The panel's preferred DMT 39 (1360x768) mode is non-standard and
# was producing intermittent flicker that the signal-strength knobs
# above didn't fix. CEA timings are part of every HDMI receiver's
# baseline support and lock reliably. The monitor will scale 720p
# to its native 1360x768 internally.
#
# When this hypothesis is no longer needed (Newton ships, panel
# choice is finalised, etc.) revisit: we lose some pixels and
# add a scaling step, both fine for now.
hdmi_group=1
hdmi_mode=4
