# Phase-0 bring-up config for the Pi Zero 2 W.
#
# References (see docs/REAL_HW_BRINGUP.md for the verified-facts section):
#   - raspberrypi.com/documentation/computers/config_txt.html
#   - raspberrypi/firmware/boot/overlays/README (master)
#   - raspberrypi/tools/armstubs/armstub8.S
#
# Trimmed against a working Raspbian Bookworm boot on the same
# Pi Zero 2 W + HDMI panel (see commit message for the dump). Removed
# settings that didn't address the firmware-stage flicker / white-bar:
#   hdmi_drive=2  hdmi_force_hotplug=1  config_hdmi_boost=7
#   hdmi_group=1  hdmi_mode=4           avoid_warnings=1
# Raspbian shows the same flicker until KMS does a clean modeset
# partway through systemd boot, so it isn't a config-knob fix; the
# remediation lives in our own re-modeset path (or its absence).

# 64-bit kernel. Selects the AArch64 armstub which eret's to EL2h before
# branching to kernel8.img at 0x80000. Without this we land at EL3 or
# the 32-bit armstub.
arm_64bit=1

# Use the cut-down GPU firmware (start_cd.elf + fixup_cd.dat).
# Per the config.txt docs: "Cut-down firmware (start*cd.elf and
# fixup*cd.dat) cannot be selected this way [via start_file=] - the
# system will fail to boot. The only way to enable the cut-down
# firmware is to specify gpu_mem=16." We ship start_cd.elf, so this
# line is mandatory — otherwise the firmware looks for start.elf
# (which isn't on the SD) and halts after reading config.txt.
#
# We tested the full firmware (gpu_mem=64, ships start.elf): it did
# not clear the persistent top-row white bar, so we stayed on the
# cut-down variant. See `host_io::pi_fb::FIRMWARE_TOP_BAR_PX` for
# the workaround.
gpu_mem=16

# Suppress the firmware's rainbow boot splash. Our own boot splash
# takes over from the moment `display::splash::init` runs in kmain.
disable_splash=1

# Our binary (built by scripts/build-sd.sh from the pi-probe [[bin]]).
kernel=kernel8.img

# Enable a serial console on GPIO 14/15 (header pins 8 and 10).
enable_uart=1

# Also enable firmware-side UART logging from bootcode.bin and start.elf.
# Useful as an early checkpoint: if no bytes appear on the wire at all,
# the problem is upstream of our kernel (SD layout, GPIO ALT, baud).
uart_2ndstage=1

# CRITICAL: PL011 (UART0) is wired to the onboard Bluetooth chip by
# default on the Pi Zero 2 W; the GPIO header carries the mini-UART
# (UART1) instead. Our code drives PL011 at 0x3F20_1000, so we route it
# to GPIO 14/15 by disabling BT. Quote from raspberrypi/firmware
# overlays/README: "Disable onboard Bluetooth on Bluetooth-capable
# Raspberry Pis. On Pis prior to Pi 5 this restores UART0/ttyAMA0 over
# GPIOs 14 & 15."
dtoverlay=disable-bt

# Disable HDMI overscan padding. Matches Raspbian. Some monitors trim
# a fixed margin assuming consumer-TV overscan; on monitors with no
# real overscan that margin is just lost pixels.
disable_overscan=1

# Probe HDMI EDID and pick the panel's preferred mode. Matches the
# Raspbian default; replaces the explicit `hdmi_group=1 hdmi_mode=4`
# (CEA 720p60) we used as a flicker workaround. We render at panel
# native resolution and choose Newton's screen geometry as
# `panel / 2` so each Newton pixel is a 2×2 panel block — no panel-
# side scaling, no resampling. See `host_io::pi_fb::init`.
display_auto_detect=1
